## `crtm/aggregation.py`

```python
"""Non-compensatory aggregation operators (Section 6).

All operators take a vector m in [0, 1]^q of normalized metrics, a weight
vector w on the simplex and a set K of core (knockout) positions
(0-based indices).

  additive     S_w(m)                                  (weighted mean)
  hard_gate    mu^Phi   = Phi_K(m) S_w(m)              (eq. 14)
  power        mu^(g)   = (prod_K m_k)^g S_w(m)        (eq. 15)
  capped       mu^[th]  = min{S_w, min_K m_k / th_k}   (eq. 16)
  missing      mu_O and the enclosure of Proposition 6 (eqs. 17-18)
"""
from __future__ import annotations

from typing import Iterable, Sequence

import numpy as np


def _w(w, q: int | None = None) -> np.ndarray:
    w = np.asarray(w, dtype=float)
    if np.any(w < 0) or not np.isclose(w.sum(), 1.0, atol=1e-9):
        raise ValueError("weights must be nonnegative and sum to 1")
    if q is not None and w.size != q:
        raise ValueError("weights and metrics differ in length")
    return w


def _theta(theta, K: Sequence[int]) -> np.ndarray:
    th = np.broadcast_to(np.asarray(theta, dtype=float), (len(K),))
    if np.any(th <= 0) or np.any(th > 1):
        raise ValueError("thresholds theta_k must lie in (0, 1]")
    return th


def additive(m, w) -> float:
    """Weighted arithmetic mean S_w(m)."""
    m = np.asarray(m, dtype=float)
    return float(_w(w, m.size) @ m)


def gate(m, K: Iterable[int]) -> int:
    """Stop factor Phi_K(m) = prod_{k in K} 1[m_k > 0]."""
    m = np.asarray(m, dtype=float)
    return int(all(m[k] > 0 for k in K))


def hard_gate(m, w, K: Iterable[int] = ()) -> float:
    """Hard-gated score mu^Phi (eq. 14)."""
    return gate(m, K) * additive(m, w)


def power(m, w, K: Sequence[int], gamma: float = 1.0) -> float:
    """Power-discounted score mu^(gamma) (eq. 15)."""
    m = np.asarray(m, dtype=float)
    core = np.prod([m[k] for k in K]) if len(K) else 1.0
    return float(core ** gamma * additive(m, w))


def capped(m, w, K: Sequence[int], theta=0.5) -> float:
    """Capped knockout operator mu^[theta] (Definition 9, eq. 16)."""
    m = np.asarray(m, dtype=float)
    s = additive(m, w)
    if not len(K):
        return s
    th = _theta(theta, K)
    return float(min(s, min(m[k] / t for k, t in zip(K, th))))


def compensation_capacity_additive(w, k: int) -> float:
    """C_k(S_w) = 1 - w_k (Section 6.1)."""
    return float(1.0 - np.asarray(w, dtype=float)[k])


# ------------------------------------------------------------ missing data
def missing_enclosure(m, w, K: Sequence[int] = (), theta=None) -> dict:
    """Renormalized score and enclosure for missing metrics (Proposition 6).

    Missing metrics are NaN in ``m``. Returns a dict with completeness kappa,
    renormalized score mu_O (eq. 17), lower/upper bound of (18), and the
    status. A missing core metric gives status 'not computed'.
    """
    m = np.asarray(m, dtype=float)
    w = _w(w, m.size)
    obs = ~np.isnan(m)
    if any(not obs[k] for k in K):
        return {"status": "not computed: missing core metric", "kappa": float(w[obs].sum()),
                "mu_O": np.nan, "lower": np.nan, "upper": np.nan}
    kappa = float(w[obs].sum())
    if kappa <= 0:
        raise ValueError("no observed metric carries weight")
    g = gate(np.where(obs, m, 1.0), K)
    mu_o = g * float(w[obs] @ m[obs]) / kappa
    lower = kappa * mu_o
    upper = kappa * mu_o + (1 - kappa) if g else 0.0
    if theta is not None and len(K):
        th = _theta(theta, K)
        cap = min(m[k] / t for k, t in zip(K, th))
        lower, upper, mu_o_cap = min(lower, cap), min(upper, cap), min(mu_o, cap)
    else:
        mu_o_cap = mu_o
    return {"status": "ok" if kappa == 1 else "enclosure", "kappa": kappa,
            "mu_O": float(mu_o_cap), "lower": float(lower), "upper": float(upper),
            "max_deviation": float((1 - kappa) * max(mu_o, 1 - mu_o)) if g else 0.0}


# ---------------------------------------------------------- property tests
def is_idempotent(op, q: int, ts=np.linspace(0, 1, 101)) -> bool:
    """Numerically test A(t, ..., t) = t."""
    return bool(all(np.isclose(op(np.full(q, t)), t) for t in ts))


def has_veto(op, q: int, k: int, trials: int = 200, seed: int = 0) -> bool:
    """Numerically test m_k = 0 => A(m) = 0 on random points."""
    rng = np.random.default_rng(seed)
    for _ in range(trials):
        m = rng.random(q)
        m[k] = 0.0
        if op(m) != 0.0:
            return False
    return True


def max_jump_at(op, m, k: int, eps: float = 1e-9) -> float:
    """|A(m with m_k = eps) - A(m with m_k = 0)|: a continuity probe."""
    a = np.array(m, dtype=float)
    b = a.copy()
    a[k], b[k] = 0.0, eps
    return abs(op(b) - op(a))
```

## `crtm/evaluation.py`

```python
"""Evaluation algorithm for one threat scenario (Section 6.7, Figure 8)
and roll-up to a resilience profile (Definition 10)."""
from __future__ import annotations

import math
from dataclasses import dataclass, field

import numpy as np

from . import aggregation as agg
from .metrics import Action
from .scale import level, level_abbr

OPERATORS = ("additive", "hard_gate", "power", "capped")


@dataclass
class ActionResult:
    ident: str
    goal: str
    status: str
    score: float
    level: int
    kappa: float
    lower: float
    upper: float
    contributions: list[float] = field(default_factory=list)
    normalized: list[float] = field(default_factory=list)
    inconsistent: list[int] = field(default_factory=list)

    def as_dict(self) -> dict:
        d = self.__dict__.copy()
        d["level_abbr"] = level_abbr(self.score)
        return d


def score_vector(m, w, K, operator: str = "hard_gate", gamma: float = 1.0, theta=0.5) -> float:
    """Apply one of the action operators to a complete vector."""
    if operator == "additive":
        return agg.additive(m, w)
    if operator == "hard_gate":
        return agg.hard_gate(m, w, K)
    if operator == "power":
        return agg.power(m, w, K, gamma)
    if operator == "capped":
        return agg.capped(m, w, K, theta)
    raise ValueError(f"unknown operator {operator!r}")


def evaluate_action(action: Action, method: str = "anchored", operator: str = "hard_gate",
                    gamma: float = 1.0, theta=0.5) -> ActionResult:
    """Score one basic action following Figure 8.

    1. validate class tables (8) and weights;
    2. refuse to score if a core (knockout) metric is missing;
    3. normalize each observed metric once;
    4. apply the operator; if metrics are missing, return mu_O with the
       enclosure of Proposition 6.
    """
    for mt in action.metrics:
        mt.table.validate()
    w = action.w()
    agg._w(w, action.q)
    K = action.core
    m = action.vector(method)
    inconsistent = [p for p, mt in enumerate(action.metrics)
                    if mt.observed and level(mt.normalized(method)) != mt.tabulated_level()]
    if any(math.isnan(m[k]) for k in K):
        return ActionResult(action.ident, action.goal, "not computed: missing core metric",
                            math.nan, 0, float(w[~np.isnan(m)].sum()), math.nan, math.nan,
                            normalized=m.tolist(), inconsistent=inconsistent)
    contrib = [float(w[p] * m[p]) if not math.isnan(m[p]) else math.nan for p in range(action.q)]
    if not np.isnan(m).any():
        s = score_vector(m, w, K, operator, gamma, theta)
        return ActionResult(action.ident, action.goal, "ok", s, level(s), 1.0, s, s,
                            contrib, m.tolist(), inconsistent)
    if operator not in ("hard_gate", "capped", "additive"):
        raise ValueError("the enclosure of Proposition 6 is stated for additive, gate and cap")
    enc = agg.missing_enclosure(m, w, K if operator != "additive" else (),
                                theta if operator == "capped" else None)
    return ActionResult(action.ident, action.goal, enc["status"], enc["mu_O"], level(enc["mu_O"]),
                        enc["kappa"], enc["lower"], enc["upper"], contrib, m.tolist(), inconsistent)


def evaluate_scenario(actions: dict[str, Action], selected=None, method: str = "anchored",
                      operator: str = "hard_gate", **kw) -> dict:
    """Evaluate the selected actions and roll up to goals (mean and minimum)."""
    selected = list(selected) if selected is not None else list(actions)
    res = {b: evaluate_action(actions[b], method, operator, **kw) for b in selected}
    goals: dict[str, list[float]] = {}
    for r in res.values():
        if not math.isnan(r.score):
            goals.setdefault(r.goal, []).append(r.score)
    profile = {g: {"mean": float(np.mean(v)), "min": float(np.min(v)),
                   "level_mean": level_abbr(np.mean(v)), "level_min": level_abbr(np.min(v))}
               for g, v in goals.items()}
    n_metrics = sum(actions[b].q for b in selected)
    return {"actions": res, "profile": profile, "N_theta": n_metrics,
            "observed": sum(mt.observed for b in selected for mt in actions[b].metrics)}


def dominates(p: dict, q: dict, key: str = "mean") -> bool:
    """Product order on profiles: p >= q in every goal."""
    return all(p[g][key] >= q[g][key] for g in q)
```

## `crtm/hierarchy.py`

```python
"""Set-theoretic data model (Section 3).

  * identifier map iota(i, j, k) = c_i.Sj.Ak and its inverse (Proposition 1);
  * technique alphabet with an injective coding (Definition 2);
  * scenario projections, admissibility (3), order, join and meet (4)
    and the down-set encoding of Proposition 2;
  * Jaccard distance between projections (Section 3.3).
"""
from __future__ import annotations

import re
import unicodedata
from dataclasses import dataclass, field
from typing import Iterable, Mapping

#: CREF tasks (objectives) in index order i = 1..8
TASKS = {1: ("PA", "Prevent/Avoid"), 2: ("PR", "Prepare"), 3: ("CN", "Continue"),
         4: ("CS", "Constrain"), 5: ("RE", "Reconstitute"), 6: ("UN", "Understand"),
         7: ("TR", "Transform"), 8: ("RA", "Re-Architect")}

#: CREF goals
GOALS = {"ANT": "Anticipate", "WITH": "Withstand", "REC": "Recover", "AD": "Adapt"}

#: the fourteen techniques of NIST SP 800-160 Vol. 2 with disambiguated codes
TECHNIQUES = {"AR": "Adaptive Response", "AM": "Analytic Monitoring",
              "CA": "Contextual Awareness", "CP": "Coordinated Protection",
              "DE": "Deception", "DV": "Diversity", "DP": "Dynamic Positioning",
              "NP": "Non-Persistence", "PR": "Privilege Restriction",
              "RD": "Redundancy", "RL": "Realignment", "S": "Segmentation",
              "SI": "Substantiated Integrity", "U": "Unpredictability"}

_CODE_RE = re.compile(r"^[A-Z]+$")
_ID_RE = re.compile(r"^([A-Z]+)\.S([1-9][0-9]*)\.A([1-9][0-9]*)$")
_TASK_BY_CODE = {c: i for i, (c, _) in TASKS.items()}


# --------------------------------------------------------------- identifiers
def check_code(code: str) -> list[str]:
    """Return problems with a task/technique code (empty list if valid).

    Detects non-Latin look-alike letters, e.g. a Cyrillic 'С' in 'СN'
    (one such entry was found in the source tabulation, Section 3.4).
    """
    problems = []
    for ch in code:
        if not ("A" <= ch <= "Z"):
            name = unicodedata.name(ch, "UNKNOWN")
            problems.append(f"character {ch!r} (U+{ord(ch):04X}, {name}) is not a Latin capital")
    if not code:
        problems.append("empty code")
    return problems


def iota(i: int, j: int, k: int, tasks: Mapping[int, tuple[str, str]] = TASKS) -> str:
    """Identifier map iota(i, j, k) = c_i.Sj.Ak (Definition 1)."""
    if min(i, j, k) < 1:
        raise ValueError("indices start at 1")
    return f"{tasks[i][0]}.S{j}.A{k}"


def iota_inv(identifier: str, tasks: Mapping[int, tuple[str, str]] = TASKS) -> tuple[int, int, int]:
    """Inverse of iota by parsing (Proposition 1)."""
    mt = _ID_RE.match(identifier)
    if not mt:
        bad = check_code(identifier.split(".")[0])
        raise ValueError(f"malformed identifier {identifier!r}" + (f": {bad}" if bad else ""))
    by_code = {c: i for i, (c, _) in tasks.items()}
    code, j, k = mt.group(1), int(mt.group(2)), int(mt.group(3))
    if code not in by_code:
        raise ValueError(f"unknown task code {code!r}")
    return by_code[code], j, k


def is_injective(mapping: Mapping) -> bool:
    """True if no two keys share a value."""
    vals = list(mapping.values())
    return len(vals) == len(set(vals))


# --------------------------------------------------------------- projections
@dataclass(frozen=True)
class Projection:
    """Projection P = (I_P, alpha_P) of the data model (Definition 3)."""

    alpha: Mapping[str, frozenset] = field(default_factory=dict)
    name: str = ""

    def __post_init__(self):
        object.__setattr__(self, "alpha",
                           {b: frozenset(a) for b, a in dict(self.alpha).items()})

    @property
    def actions(self) -> frozenset:
        return frozenset(self.alpha)

    def techniques(self, beta: str) -> frozenset:
        return self.alpha.get(beta, frozenset())

    # admissibility (3) -------------------------------------------------
    def violations(self, catalog: Mapping[str, Iterable[str]]) -> list[str]:
        out = []
        for b, a in self.alpha.items():
            if b not in catalog:
                out.append(f"{b}: not in the catalog index set")
            elif not a <= frozenset(catalog[b]):
                extra = sorted(a - frozenset(catalog[b]))
                out.append(f"{b}: techniques {extra} not in catalog set {sorted(catalog[b])}")
        return out

    def is_admissible(self, catalog: Mapping[str, Iterable[str]]) -> bool:
        return not self.violations(catalog)

    # order and lattice operations (4) ----------------------------------
    def __le__(self, other: "Projection") -> bool:
        return all(b in other.alpha and a <= other.alpha[b] for b, a in self.alpha.items())

    def __or__(self, other: "Projection") -> "Projection":   # join
        keys = self.actions | other.actions
        return Projection({b: self.techniques(b) | other.techniques(b) for b in keys},
                          name=f"({self.name} ⊔ {other.name})")

    def __and__(self, other: "Projection") -> "Projection":  # meet
        keys = self.actions & other.actions
        return Projection({b: self.techniques(b) & other.techniques(b) for b in keys},
                          name=f"({self.name} ⊓ {other.name})")

    def __eq__(self, other) -> bool:
        return isinstance(other, Projection) and dict(self.alpha) == dict(other.alpha)

    def __hash__(self):
        return hash(frozenset(self.alpha.items()))

    # down-set encoding (proof of Proposition 2) ------------------------
    def downset(self) -> frozenset:
        """E(P) = {(beta, *)} ∪ {(beta, a) : a in alpha_P(beta)}."""
        return frozenset([(b, "*") for b in self.alpha] +
                         [(b, a) for b, s in self.alpha.items() for a in s])


def poset_size(catalog: Mapping[str, Iterable[str]]) -> int:
    """|U| = |I| + sum |alpha(beta)| (Proposition 2)."""
    return len(catalog) + sum(len(set(a)) for a in catalog.values())


def jaccard_index(a: frozenset, b: frozenset) -> float:
    union = a | b
    return 1.0 if not union else len(a & b) / len(union)


def jaccard_distance(P: Projection, Q: Projection, on: str = "actions") -> float:
    """d_J on action sets (on='actions') or on down-sets (on='downset')."""
    if on == "actions":
        return 1.0 - jaccard_index(P.actions, Q.actions)
    if on == "downset":
        return 1.0 - jaccard_index(P.downset(), Q.downset())
    raise ValueError("on must be 'actions' or 'downset'")
```

```python
"""Metric model (Section 4) and loaders for the supplementary data files."""
from __future__ import annotations

import csv

@dataclass
class Metric:
    """Metric descriptor M = (X, u, sigma, tau, A, xi) with an observed value (Definition 4)."""

    code: str
    name: str
    unit: str
    table: ClassTable
    value: float = math.nan
    core: bool = False
    timing: str = ""            # 'ex-ante', 'ex-post' or '' (unspecified, ⊥)
    printed_classes: str = ""
    support: frozenset = frozenset()

    @property
    def observed(self) -> bool:
        return not math.isnan(self.value)

    @property
    def sigma(self) -> int:
        return self.table.sigma

    def normalized(self, method: str = "anchored") -> float:
        return math.nan if not self.observed else float(self.table.normalize(self.value, method))

    def tabulated_level(self) -> int:
        return 0 if not self.observed else int(self.table.tabulated_level(self.value))


@dataclass
class Action:
    """Basic action beta with its ordered metric tuple M(beta) and weights."""

    ident: str
    goal: str
    metrics: list[Metric] = field(default_factory=list)
    weights: np.ndarray | None = None
    weights_specified: bool = False

    @property
    def q(self) -> int:
        return len(self.metrics)

    @property
    def core(self) -> list[int]:
        return [p for p, m in enumerate(self.metrics) if m.core]

    def w(self) -> np.ndarray:
        if self.weights is None:
            return np.full(self.q, 1.0 / self.q)
        return np.asarray(self.weights, dtype=float)

    def vector(self, method: str = "anchored") -> np.ndarray:
        return np.array([m.normalized(method) for m in self.metrics])

    def check_support(self, alpha: frozenset) -> list[str]:
        """Support condition A_p ⊆ alpha(beta) of Definition 4."""
        return [f"{self.ident} p={p + 1}: support {sorted(m.support)} ⊄ {sorted(alpha)}"
                for p, m in enumerate(self.metrics) if not m.support <= alpha]


def _f(s: str) -> float:
    return math.nan if s.strip() == "" else float(s)


def load_metrics(path: str | Path | None = None) -> dict[str, Action]:
    """Load ``phishing_metrics.csv`` into an ordered dict of actions."""
    path = Path(path) if path else DATA_DIR / "phishing_metrics.csv"
    actions: dict[str, Action] = {}
    raw_w: dict[str, list[float]] = {}
    with open(path, encoding="utf-8", newline="") as fh:
        for row in csv.DictReader(fh):
            a = actions.setdefault(row["action"], Action(row["action"], row["goal"]))
            xi = tuple(float(row[f"xi{r}"]) for r in range(6))
            a.metrics.append(Metric(code=row["code"], name=row["metric"], unit=row["unit"],
                                    table=ClassTable(xi), value=_f(row["value"]),
                                    core=row["core"].strip() == "1", timing=row["timing"],
                                    printed_classes=row["printed_classes"]))
            raw_w.setdefault(a.ident, []).append(_f(row["weight"]))
    for a in actions.values():
        w = raw_w[a.ident]
        if all(not math.isnan(v) for v in w):
            a.weights, a.weights_specified = np.array(w), True
            if not math.isclose(sum(w), 1.0, abs_tol=1e-9):
                raise ValueError(f"weights of {a.ident} do not sum to 1")
    return actions
```

## `crtm/normalization.py`

```python
"""Class tables and level-consistent normalization (Section 5).

Implements
  * Definition 6  – class table xi, quality anchors eta = sigma * xi,
                    tabulated level lambda_xi (eq. 9);
  * eq. (10)      – two-anchor linear normalization n_{c,h};
  * Theorem 1     – existence test for a consistent clipped affine map;
  * Corollary 1   – interval of class-4 values sent to the band "high";
  * Definition 7  – anchored piecewise-linear normalization nu_xi (eq. 12);
  * a numerical grid check of consistency (Section 7.1).
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Callable, Sequence

import numpy as np

from .scale import Y, level


class ClassTableError(ValueError):
    """Raised when a class table violates the ordering condition (8)."""


@dataclass(frozen=True)
class ClassTable:
    """Class table xi in R^6 of a metric (Definition 6).

    Parameters
    ----------
    xi : six anchors (xi_0, ..., xi_5) in physical units.
    sigma : polarity, +1 if larger values mean more resilience, -1 otherwise.
        If omitted it is inferred from the order of xi_0 and xi_4.
    """

    xi: tuple[float, ...]
    sigma: int | None = None
    eta: np.ndarray = field(init=False, repr=False, compare=False)

    def __post_init__(self):
        xi = tuple(float(v) for v in self.xi)
        if len(xi) != 6:
            raise ClassTableError("a class table has exactly six anchors")
        sigma = self.sigma
        if sigma is None:
            sigma = 1 if xi[4] > xi[0] else -1
        if sigma not in (1, -1):
            raise ClassTableError("polarity must be +1 or -1")
        eta = sigma * np.asarray(xi)
        object.__setattr__(self, "xi", xi)
        object.__setattr__(self, "sigma", int(sigma))
        object.__setattr__(self, "eta", eta)
        self.validate()

    # ------------------------------------------------------------------ checks
    def validate(self) -> None:
        """Check condition (8): eta0 <= eta1 < eta2 < eta3 < eta4 <= eta5."""
        e = self.eta
        ok = e[0] <= e[1] < e[2] < e[3] < e[4] <= e[5]
        if not ok:
            raise ClassTableError(
                f"anchors violate (8): eta = {tuple(np.round(e, 6))}")

    # --------------------------------------------------------- tabulated level
    def tabulated_level(self, x):
        """Level lambda_xi(x) that the class table assigns to x (eq. 9)."""
        u = self.sigma * np.asarray(x, dtype=float)
        e = self.eta
        r = np.searchsorted(e[1:5], u, side="right") + 1
        r = np.where(np.isnan(u), 0, r)
        return int(r) if np.ndim(r) == 0 else r.astype(int)

    # ---------------------------------------------------------- normalizations
    def linear(self, x):
        """Two-anchor linear normalization n_{c,h} with c = xi_0, h = xi_4 (eq. 10)."""
        c, h = self.xi[0], self.xi[4]
        out = np.clip((np.asarray(x, dtype=float) - c) / (h - c), 0.0, 1.0)
        return float(out) if out.ndim == 0 else out

    def anchored(self, x):
        """Anchored piecewise-linear normalization nu_xi (Definition 7, eq. 12).

        Degenerate segments (eta_r = eta_{r+1}, possible for r = 0 and r = 4)
        are skipped. At a degenerate critical anchor eta_0 = eta_1 the value
        at u = eta_1 is y_1, so that nu_xi is consistent with xi on all of R
        (Theorem 2a).
        """
        u = self.sigma * np.atleast_1d(np.asarray(x, dtype=float))
        e = self.eta
        out = np.full(u.shape, np.nan)
        out[u <= e[0]] = 0.0
        for r in range(5):
            if e[r] < e[r + 1]:
                m = (u >= e[r]) & (u < e[r + 1])
                out[m] = Y[r] + (Y[r + 1] - Y[r]) * (u[m] - e[r]) / (e[r + 1] - e[r])
        out[u >= e[5]] = 1.0
        if e[4] == e[5]:
            out[u >= e[4]] = 1.0
        out[np.isnan(u)] = np.nan
        return float(out[0]) if np.ndim(x) == 0 else out

    def normalize(self, x, method: str = "anchored"):
        """Dispatch to ``linear`` or ``anchored``."""
        if method == "linear":
            return self.linear(x)
        if method == "anchored":
            return self.anchored(x)
        raise ValueError(f"unknown normalization {method!r}")

    # ---------------------------------------------------------- theory checks
    def affine_consistent(self, rtol: float = 1e-9) -> tuple[bool, float | None, float | None]:
        """Theorem 1: does a consistent clipped affine map a*u + b exist?

        Returns (exists, a, b). Condition (11): equal consecutive slopes
        (eta_{r+1} - eta_r) / (y_{r+1} - y_r) for r = 1, 2, 3.
        """
        e = self.eta
        slopes = [(e[r + 1] - e[r]) / (Y[r + 1] - Y[r]) for r in (1, 2, 3)]
        exists = bool(np.allclose(slopes, slopes[0], rtol=rtol, atol=0.0))
        if not exists:
            return False, None, None
        a = 1.0 / slopes[0]
        b = Y[1] - a * e[1]
        return True, float(a), float(b)

    def corollary1_interval(self) -> dict:
        """Corollary 1: class-4 values that n_{c,h} maps into the band 'high'.

        Returns the interval in quality coordinates u = sigma*x, the same
        interval in physical units and its length
        min{eta4 - eta3, (1 - y4)(eta4 - eta0)}.
        """
        e = self.eta
        lo = max(e[3], e[0] + Y[4] * (e[4] - e[0]))
        hi = e[4]
        length = min(e[4] - e[3], (1 - Y[4]) * (e[4] - e[0]))
        if self.sigma == 1:
            phys = {"from": float(lo), "to": float(hi), "closed": "[from, to)"}
        else:
            phys = {"from": float(-hi), "to": float(-lo), "closed": "(from, to]"}
        return {"u_interval": (float(lo), float(hi)), "x_interval": phys,
                "length": float(length)}

    def anchors_with_neighbours(self, eps: float = 1e-9) -> np.ndarray:
        """Physical anchors and points within eps of them (for grid checks)."""
        pts = []
        for v in self.xi:
            pts += [v - eps, v, v + eps]
        return np.asarray(pts)


# ---------------------------------------------------------------- utilities
def is_consistent(table: ClassTable, norm: Callable, xs: Sequence[float]) -> np.ndarray:
    """Boolean array: lambda(n(x)) == lambda_xi(x) for each x in xs."""
    xs = np.asarray(xs, dtype=float)
    return level(norm(xs)) == table.tabulated_level(xs)


def grid_check(table: ClassTable, method: str = "anchored", n: int = 20001,
               margin: float = 0.25, eps: float = 1e-9) -> dict:
    """Numerical consistency check on a dense grid plus all anchors ± eps.

    The grid covers [min xi - margin*span, max xi + margin*span].
    """
    lo, hi = min(table.xi), max(table.xi)
    span = hi - lo
    xs = np.concatenate([np.linspace(lo - margin * span, hi + margin * span, n),
                         table.anchors_with_neighbours(eps)])
    ok = is_consistent(table, lambda v: table.normalize(v, method), xs)
    return {"points": int(xs.size), "violations": int((~ok).sum()),
            "worst_x": xs[~ok].tolist()[:10]}
```

## `crtm/scale.py`

```python
"""Five-level linguistic scale (Definition 5).

Score breakpoints y = (0, 0.2, 0.4, 0.7, 0.9, 1) define the bands
Y_r = [y_{r-1}, y_r) for r = 1..4 and Y_5 = [y_4, y_5], named
critical, below average, average, above average and high.
"""
from __future__ import annotations

import numpy as np

#: score breakpoints y_0 .. y_5
Y = np.array([0.0, 0.2, 0.4, 0.7, 0.9, 1.0])

#: level names, index r - 1 for r = 1..5
LEVEL_NAMES = ("critical", "below average", "average", "above average", "high")

#: abbreviations used in the article tables
LEVEL_ABBR = ("C", "BA", "A", "AA", "H")


def level(s):
    """Level map lambda: [0, 1] -> {1, ..., 5} (vectorised).

    Returns an int (scalar input) or an int array. NaN maps to 0
    ("not computed").
    """
    s_arr = np.asarray(s, dtype=float)
    r = np.searchsorted(Y[1:5], s_arr, side="right") + 1
    r = np.where(np.isnan(s_arr), 0, r)
    if r.ndim == 0:
        return int(r)
    return r.astype(int)


def level_abbr(s) -> str:
    """Abbreviated level name of a single score, '—' for NaN."""
    r = level(s)
    return "—" if r == 0 else LEVEL_ABBR[r - 1]


def level_name(s) -> str:
    """Full level name of a single score."""
    r = level(s)
    return "not computed" if r == 0 else LEVEL_NAMES[r - 1]


def band(r: int) -> tuple[float, float]:
    """Score band (y_{r-1}, y_r) of level r."""
    if not 1 <= r <= 5:
        raise ValueError("level must be in 1..5")
    return float(Y[r - 1]), float(Y[r])
```

```python
"""Sensitivity of an action score to two response times (Section 7.4)."""
from __future__ import annotations

import numpy as np
from scipy.optimize import brentq

from . import aggregation as agg
from .metrics import Action


def score_with(action: Action, values: dict[int, float], method: str) -> float:
    """Hard-gated score of ``action`` with metric values replaced (0-based positions)."""
    m = []
    for p, mt in enumerate(action.metrics):
        x = values.get(p, mt.value)
        m.append(float(mt.table.normalize(x, method)))
    return agg.hard_gate(np.array(m), action.w(), action.core)


def grid(action: Action, p1: int, p2: int, r1, r2, method: str, n1: int = 401, n2: int = 521):
    """Score on a (n2 x n1) grid over metric p1 (x-axis) and p2 (y-axis)."""
    xs, ys = np.linspace(*r1, n1), np.linspace(*r2, n2)
    t1 = action.metrics[p1].table.normalize(xs, method)
    t2 = action.metrics[p2].table.normalize(ys, method)
    base = action.vector(method)
    w = action.w()
    rest = base @ w - w[p1] * base[p1] - w[p2] * base[p2]
    Z = rest + w[p1] * t1[None, :] + w[p2] * t2[:, None]
    return xs, ys, Z


def threshold(action: Action, p: int, target: float, method: str, lo: float, hi: float,
              fixed: dict[int, float] | None = None) -> float:
    """Value of metric p at which the score crosses ``target`` (root in [lo, hi])."""
    fixed = dict(fixed or {})
    f = lambda x: score_with(action, {**fixed, p: x}, method) - target
    return float(brentq(f, lo, hi, xtol=1e-10))
```

## `crtm/uncertainty.py`

```python
"""Uncertainty in the weights (Section 6.5, Proposition 7)."""
from __future__ import annotations

import numpy as np

from .scale import Y, level

SEED = 20261027          # fixed seed used in the article
N_SAMPLES = 200_000      # weight vectors per configuration


def dirichlet_moments(m, alpha: float) -> tuple[float, float]:
    """Exact mean and standard deviation of S_w(m) for w ~ Dirichlet(alpha 1_q) (eq. 19)."""
    m = np.asarray(m, dtype=float)
    q = m.size
    s2 = np.mean((m - m.mean()) ** 2)
    return float(m.mean()), float(np.sqrt(s2 / (q * alpha + 1)))


def sample_scores(m, alpha: float, n: int = N_SAMPLES, seed: int = SEED) -> np.ndarray:
    """Monte Carlo sample of S_w(m) under symmetric Dirichlet weights."""
    m = np.asarray(m, dtype=float)
    rng = np.random.default_rng(seed)
    w = rng.dirichlet(np.full(m.size, alpha), size=n)
    return w @ m


def level_probabilities(scores) -> dict[int, float]:
    """Empirical Pr[lambda(S) = r], r = 1..5."""
    lv = level(scores)
    return {r: float(np.mean(lv == r)) for r in range(1, 6)}


def summary(m, alpha: float, n: int = N_SAMPLES, seed: int = SEED) -> dict:
    s = sample_scores(m, alpha, n, seed)
    mean, sd = dirichlet_moments(m, alpha)
    return {"alpha": alpha, "exact_mean": mean, "exact_sd": sd,
            "mc_mean": float(s.mean()), "mc_sd": float(s.std()),
            "p_high": float(np.mean(s >= Y[4])), "p_below_0.7": float(np.mean(s < Y[3])),
            "levels": level_probabilities(s)}
```

```python
FONT_DIRS = [Path("/usr/share/fonts/truetype/dejavu"), Path("/usr/share/fonts/dejavu"),
             Path("/Library/Fonts"), Path("C:/Windows/Fonts")]


def _register_fonts() -> tuple[str, str]:
    for d in FONT_DIRS:
        mono, sans = d / "DejaVuSansMono.ttf", d / "DejaVuSans.ttf"
        if mono.exists() and sans.exists():
            pdfmetrics.registerFont(TTFont("Mono", str(mono)))
            pdfmetrics.registerFont(TTFont("Sans", str(sans)))
            bold = d / "DejaVuSans-Bold.ttf"
            pdfmetrics.registerFont(TTFont("SansB", str(bold if bold.exists() else sans)))
            return "Mono", "Sans"
    raise SystemExit("DejaVu fonts not found (needed for Greek and math symbols)")


def sha(p: Path) -> str:
    return hashlib.sha256(p.read_bytes()).hexdigest()


def build_pdf(out: Path):

        c.setFont(sans, 7)
        c.setFillGray(0.4)
        c.drawString(left, 9 * mm, f"crtm {__version__} — program code listing")
        c.drawRightString(W - 15 * mm, 9 * mm, f"page {page[0]}")
        c.setFillGray(0)

    def new_page():
        footer()
        c.showPage()
        page[0] += 1

    c.setFont("SansB", 15)
    c.drawString(left, top - 10 * mm, "CRTM — Cyber-Resilience Tuple Model")
    c.setFont(sans, 11)
    c.drawString(left, top - 18 * mm, f"Program code listing, software version {__version__}")

    lines = ["Digital supplement (Code S1) to the article “A Set-Theoretic Tuple Model with Level-Consistent",

             f"Generated {dt.date.today().isoformat()} by tools/build_listing.py."]
    y = top - 28 * mm
    for s in lines:
        c.drawString(left, y, s)
        y -= 11
    y -= 10
    c.setFont("SansB", 9)
    c.drawString(left, y, "File")
    c.drawString(left + 70 * mm, y, "Lines")

                y -= lh

def build_md(out: Path):
    parts = [f"# {TITLE}\n", f"Software version {__version__}. Generated by `tools/build_listing.py`.\n",
             "| File | Lines | SHA-256 |", "|---|---:|---|"]
    for p in FILES:
        parts.append(f"| `{p.relative_to(ROOT)}` | {len(p.read_text(encoding='utf-8').splitlines())} | `{sha(p)}` |")
    for p in FILES:
        parts += ["", f"## `{p.relative_to(ROOT)}`", "", "```python", p.read_text(encoding="utf-8").rstrip(), "```"]
    out.write_text("\n".join(parts) + "\n", encoding="utf-8")
```
